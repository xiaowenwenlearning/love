        // Audio synthesizer using Web Audio API
        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        function playTone(freq, type, duration) {
            if (!soundEnabled) return;
            try {
                if (audioCtx.state === 'suspended') audioCtx.resume();
                const osc = audioCtx.createOscillator();
                const gain = audioCtx.createGain();
                osc.type = type;
                osc.frequency.value = freq;
                gain.gain.setValueAtTime(0.15, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + duration);
                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start();
                osc.stop(audioCtx.currentTime + duration);
            } catch (e) { console.log(e); }
        }

        // Handle user.txt upload
        function handleUserTxtUpload(e) {
            const file = e.target.files[0];
            if (!file) return;
            document.getElementById('user-file-name').innerText = file.name;
            const reader = new FileReader();
            reader.onload = function(event) {
                const text = event.target.result;
                const lines = text.split(/\r?\n/);
                let loadedCount = 0;
                lines.forEach(line => {
                    const parts = line.split(/\t|,|\s+/);
                    if (parts.length >= 2) {
                        const u = parts[0].trim();
                        const p = parts[1].trim();
                        if (u && p) {
                            userDatabase[u] = p;
                            loadedCount++;
                        }

                });
                const status = document.getElementById('user-status');
                status.innerText = `成功导入 ${loadedCount} 个账号！`;
                status.classList.remove('hidden');
            };
            reader.readAsText(file, 'UTF-8');
        }

        // Handle CSV Upload based on exact rules
        function handleCsvUpload(e) {
            const file = e.target.files[0];
            if (!file) return;
            document.getElementById('csv-file-name').innerText = file.name;

            Papa.parse(file, {
                complete: function(results) {
                    parseCsvData(results.data);
                },
                error: function(err) {
                    alert('CSV 解析错误: ' + err.message);
                }
            });
        }

        function parseCsvData(rows) {
            // Filter non-empty rows
            const validRows = rows.filter(r => r && r.length > 1);
            if (validRows.length < 4) {
                alert('CSV 文件行数不足，请确保至少包含4行（待选、固定、位置、答案）！');
                return;
            }

            const newLevels = [];
            // Assuming every 4 rows make a level, or parse block by block
            for (let i = 0; i <= validRows.length - 4; i += 4) {
                let r1 = validRows[i].slice(1, 11).map(s => String(s || '').trim());   // Targets (exclude col 0)
                let r2 = validRows[i+1].slice(1, 11).map(s => String(s || '').trim()); // Fixed
                let r3 = validRows[i+2].slice(1, 11).map(s => String(s || '').trim()); // Position (L, R, LR)
                let r4 = validRows[i+3].slice(1, 11).map(s => String(s || '').trim()); // Answers

                if (r1.length >= 10) {
                    newLevels.push({
                        level: newLevels.length + 1,
                        targets: r1,
                        fixed: r2,
                        positions: r3,
                        answers: r4
                    });
                }
            }

            if (newLevels.length > 0) {
                levelsData = newLevels;
                const status = document.getElementById('csv-status');
                status.innerText = `成功导入 ${newLevels.length} 个关卡！`;
                status.classList.remove('hidden');
            } else {
                alert('未能正确解析关卡数据，请检查表格结构。');
            }
        }

                    id: i,
                    target: target,
                    fixed: fixed,
                    position: pos, // L, R, LR
                    answer: ans,
                    userLeftInput: '',
                    userRightInput: '',
                    isCorrect: false
                });
            }

            // Extract candidate pool from answers + some random fillers if needed
            let rawPool = [];
            lvl.answers.forEach(ansStr => {
                // if comma separated for LR
                let chars = ansStr.replace(/,/g, '').split('');
                rawPool.push(...chars);
            });
            // Add target chars and fixed chars to pool to ensure all necessary characters are available
            lvl.targets.forEach(c => rawPool.push(c));
            lvl.fixed.forEach(c => rawPool.push(c));

            // Shuffle and make unique or keep counts
            rawPool = rawPool.filter(c => c && c.trim() !== '');
            rawPool.sort(() => Math.random() - 0.5);

            candidatePool = rawPool.map((c, idx) => ({ id: idx, char: c, used: false }));
            selectedCard = null;

            renderBoard();
            startTimer();
        }

        function renderBoard() {
            // Row 1: 10 Target characters reference
            const r1 = document.getElementById('row-1-container');
            r1.innerHTML = '';
            activeSlots.forEach((slot, idx) => {
                const div = document.createElement('div');
                div.className = 'bg-white/90 border-2 border-pink-200 rounded-2xl p-2 text-center shadow-sm flex flex-col items-center justify-center';
                div.innerHTML = `
                    <div class="text-xs text-gray-400 font-bold">#${idx+1}</div>
                    <div class="text-2xl font-fun text-dopaPink">${slot.target}</div>
                `;
                r1.appendChild(div);
            });

            // Row 2: Slots 0-4 (Left half: 5 slots)
            const r2 = document.getElementById('row-2-container');
            r2.innerHTML = '';
            for (let i = 0; i < 5; i++) {
                r2.appendChild(createSlotElement(activeSlots[i]));
            }

            // Row 3: Slots 5-9 (Right half: 5 slots)
            const r3 = document.getElementById('row-3-container');
            r3.innerHTML = '';
            for (let i = 5; i < 10; i++) {
                r3.appendChild(createSlotElement(activeSlots[i]));
            }

            // Counters
            const c1 = activeSlots.slice(0, 5).filter(s => s.isCorrect).length;
            const c2 = activeSlots.slice(5, 10).filter(s => s.isCorrect).length;
            document.getElementById('matched-counter-1').innerText = `已配对: ${c1}/5`;
            document.getElementById('matched-counter-2').innerText = `已配对: ${c2}/5`;

            // Candidate Pool
            const poolEl = document.getElementById('candidates-pool');
            poolEl.innerHTML = '';
            candidatePool.forEach((item, pIdx) => {
                const chip = document.createElement('div');
                const isSelected = selectedCard && selectedCard.id === item.id;
                chip.className = `w-12 h-12 rounded-xl flex items-center justify-center text-xl font-fun cursor-pointer transition shadow-md transform ${
                    item.used 
                        ? 'bg-gray-200 text-gray-400 opacity-40 cursor-not-allowed scale-90' 
                        : isSelected 
                            ? 'bg-gradient-to-tr from-dopaPink to-dopaYellow text-white ring-4 ring-pink-300 scale-110 animate-pulse' 
                            : 'bg-white text-gray-800 hover:scale-105 hover:bg-pink-50 border-2 border-pink-200'
                }`;
                chip.innerText = item.char;
                if (!item.used) {
                    chip.onclick = () => {
                        playClickSound();
                        selectedCard = item;
                        renderBoard();
                    };
                }
                poolEl.appendChild(chip);
            });

            lucide.createIcons();
        }

        function createSlotElement(slot) {
            const wrapper = document.createElement('div');
            wrapper.className = `p-3 rounded-2xl border-2 flex flex-col items-center justify-center transition-all ${
                slot.isCorrect ? 'bg-green-100 border-green-400 shadow-md' : 'bg-white/90 border-pink-200 shadow-sm'
            }`;

            let htmlContent = `<div class="text-xs text-gray-400 mb-1">配对槽 #${slot.id+1}</div><div class="flex items-center gap-1 text-2xl font-fun">`;

            if (slot.position === 'L') {
                // ( ) Fixed
                htmlContent += `
                    <div onclick="handleSlotClick(${slot.id}, 'left')" class="w-10 h-10 rounded-lg border-2 border-dashed border-dopaPink flex items-center justify-center bg-pink-50 cursor-pointer text-dopaPink">
                        ${slot.userLeftInput || '( )'}
                    </div>
                    <span class="text-gray-800">${slot.fixed}</span>
                `;
            } else if (slot.position === 'R') {
                // Fixed ( )
                htmlContent += `
                    <span class="text-gray-800">${slot.fixed}</span>
                    <div onclick="handleSlotClick(${slot.id}, 'right')" class="w-10 h-10 rounded-lg border-2 border-dashed border-dopaPink flex items-center justify-center bg-pink-50 cursor-pointer text-dopaPink">
                        ${slot.userRightInput || '( )'}
                    </div>
                `;
            } else { // LR
                // ( ) Fixed ( )
                htmlContent += `
                    <div onclick="handleSlotClick(${slot.id}, 'left')" class="w-9 h-9 rounded-lg border-2 border-dashed border-dopaPink flex items-center justify-center bg-pink-50 cursor-pointer text-sm text-dopaPink">
                        ${slot.userLeftInput || '()'}
                    </div>
                    <span class="text-gray-800 text-lg">${slot.fixed}</span>
                    <div onclick="handleSlotClick(${slot.id}, 'right')" class="w-9 h-9 rounded-lg border-2 border-dashed border-dopaPink flex items-center justify-center bg-pink-50 cursor-pointer text-sm text-dopaPink">
                        ${slot.userRightInput || '()'}
                    </div>
                `;
            }

            htmlContent += `</div>`;
            if (slot.userLeftInput || slot.userRightInput) {
                htmlContent += `<button onclick="clearSlot(${slot.id})" class="mt-1 text-xs text-red-500 hover:underline">清空</button>`;
            }

            wrapper.innerHTML = htmlContent;
            return wrapper;
        }

        function handleSlotClick(slotId, side) {
            if (!selectedCard) {
                playErrorSound();
                return;
            }

            const slot = activeSlots[slotId];
            if (slot.isCorrect) return;

            if (side === 'left') {
                slot.userLeftInput = selectedCard.char;
            } else {
                slot.userRightInput = selectedCard.char;
            }

            // Mark candidate as used
            selectedCard.used = true;
            selectedCard = null;

            // Check if slot answers match
            validateSlot(slot);
            renderBoard();
        }

        function clearSlot(slotId) {
            playClickSound();
            const slot = activeSlots[slotId];
            // Return characters to pool
            if (slot.userLeftInput) {
                const c = candidatePool.find(item => item.char === slot.userLeftInput && item.used);
                if (c) c.used = false;
                slot.userLeftInput = '';
            }
            if (slot.userRightInput) {
                const c = candidatePool.find(item => item.char === slot.userRightInput && item.used);
                if (c) c.used = false;
                slot.userRightInput = '';
            }
            slot.isCorrect = false;
            renderBoard();
        }

        function validateSlot(slot) {
            // Check correctness against slot.answer
            let correct = false;
            let ans = slot.answer;

            if (slot.position === 'L') {
                // answer could be single char or multiple options separated by comma
                let allowed = ans.split(',').map(s => s.trim());
                if (allowed.includes(slot.userLeftInput)) {
                    correct = true;
                }
            } else if (slot.position === 'R') {
                let allowed = ans.split(',').map(s => s.trim());
                if (allowed.includes(slot.userRightInput)) {
                    correct = true;
                }
            } else { // LR
                // format: leftAns, rightAns
                let parts = ans.split(',');
                let leftAllowed = parts[0] ? parts[0].trim().split('') : [];
                let rightAllowed = parts[1] ? parts[1].trim().split('') : [];
                if (leftAllowed.includes(slot.userLeftInput) && rightAllowed.includes(slot.userRightInput)) {
                    correct = true;
                }
            }

            if (correct) {
                playSuccessSound();
                slot.isCorrect = true;
                checkWinCondition();
            } else {
                // If both filled but incorrect
                if ((slot.position === 'L' && slot.userLeftInput) || 
                    (slot.position === 'R' && slot.userRightInput) || 
                    (slot.position === 'LR' && slot.userLeftInput && slot.userRightInput)) {
                    // Check if fully filled
                    playErrorSound();
                }
            }
        }

        function startTimer() {
            clearInterval(timerInterval);
            timeLeft = 120; // 2 minutes
            updateTimerDisplay();

            timerInterval = setInterval(() => {
                timeLeft--;
                updateTimerDisplay();

                if (timeLeft <= 0) {
                    clearInterval(timerInterval);
                    showFailModal();
                }
            }, 1000);
        }

        function updateTimerDisplay() {
            const minutes = Math.floor(timeLeft / 60);
            const seconds = timeLeft % 60;
            document.getElementById('timer-display').innerText = `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;

            const timerBox = document.getElementById('timer-box');
            if (timeLeft <= 30) {
                timerBox.classList.add('bg-red-500', 'animate-bounce');
            } else {
                timerBox.classList.remove('bg-red-500', 'animate-bounce');
            }
        }

        function checkWinCondition() {
            const allSolved = activeSlots.every(s => s.isCorrect);
            if (allSolved) {
                clearInterval(timerInterval);
                playSuccessSound();
                triggerConfetti();
                showSuccessModal();
            }
        }

        function showSuccessModal() {
            const modal = document.getElementById('success-modal');
            const timeUsed = 120 - timeLeft;
            const minUsed = Math.floor(timeUsed / 60);
            const secUsed = timeUsed % 60;
            document.getElementById('success-time').innerText = `${String(minUsed).padStart(2, '0')}:${String(secUsed).padStart(2, '0')}`;
            
            // Random cute avatar and praise
            document.getElementById('avatar-container').innerText = cuteAvatars[Math.floor(Math.random() * cuteAvatars.length)];
            document.getElementById('praise-title').innerText = praiseWords[Math.floor(Math.random() * praiseWords.length)];
            document.getElementById('praise-message').innerText = `恭喜 ${currentUser} 成功通过第 ${currentLevelIndex + 1} 关！`;

            modal.classList.remove('hidden');
            modal.classList.add('flex');
        }

        function showFailModal() {
            playErrorSound();
            const modal = document.getElementById('fail-modal');
            modal.classList.remove('hidden');
            modal.classList.add('flex');
        }

        function triggerConfetti() {
            confetti({
                particleCount: 150,
                spread: 90,
                origin: { y: 0.6 }
            });
        }

        function nextLevel() {
            document.getElementById('success-modal').classList.add('hidden');
            document.getElementById('success-modal').classList.remove('flex');
            currentLevelIndex++;
            if (currentLevelIndex >= levelsData.length) {
                currentLevelIndex = 0; // Loop or finish
            }
            initLevel(currentLevelIndex);
        }

        function restartLevel() {
            document.getElementById('fail-modal').classList.add('hidden');
            document.getElementById('fail-modal').classList.remove('flex');
            initLevel(currentLevelIndex);
        }

        function resetCurrentLevel() {
            playClickSound();
            initLevel(currentLevelIndex);
        }
